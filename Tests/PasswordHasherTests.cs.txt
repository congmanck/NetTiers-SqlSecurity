using System;
using System.Diagnostics;
using System.Security.Cryptography;
using System.Text;
using __DAL_NAMESPACE__.Security;

internal static class PasswordSecurityTests
{
    private static int checks;
    private static void Check(bool condition, string name)
    {
        if (!condition) throw new Exception("FAILED: " + name);
        checks++;
    }

    private static string Md5(string material)
    {
        using (var md5 = MD5.Create())
            return BitConverter.ToString(md5.ComputeHash(Encoding.UTF8.GetBytes(material))).Replace("-", "");
    }

    public static void Main()
    {
        bool upgrade;
        string password = "  Tiáº¿ng Viá»‡t máº­t kháº©u \ud83d\udd10 " + new string('x', 100);
        var watch = Stopwatch.StartNew();
        string stored = PasswordHasher.Hash(password);
        watch.Stop();
        Console.WriteLine("600000-iteration hash: {0} ms", watch.ElapsedMilliseconds);
        Check(PasswordHasher.IsModern(stored), "recognize production hash");
        Check(stored.Length <= 255, "storage size");
        Check(PasswordHasher.Verify("one", password, stored, false, out upgrade) && !upgrade, "Unicode and long password round trip");
        Check(PasswordHasher.Verify("renamed", password, stored, false, out upgrade), "hash independent of username");
        Check(!PasswordHasher.Verify("one", password.Trim(), stored, false, out upgrade), "do not trim modern password");
        Check(!PasswordHasher.Verify("one", "wrong", stored, false, out upgrade) && !upgrade, "wrong password");
        Check(!PasswordHasher.Verify("one", password, stored, true, out upgrade), "UIS cannot accept PBKDF2");
        string second = PasswordHasher.Hash(password);
        Check(second != stored, "independent salts");
        string local = Md5("UisPassword=" + password);
        Check(PasswordHasher.Verify("ignored", password, local.ToLowerInvariant(), false, out upgrade) && upgrade, "local legacy MD5");
        Check(!PasswordHasher.Verify("ignored", password.Trim(), local, false, out upgrade), "local legacy preserves spaces");
        string uis = Md5("UisStaffID=STAFF;UisPassword=secret");
        Check(PasswordHasher.Verify("staff", " secret ", uis, true, out upgrade) && !upgrade, "UIS legacy normalization and no migration");
        Check(!PasswordHasher.Verify("staff", "wrong", uis, true, out upgrade), "UIS wrong password");
        // Independent PBKDF2-HMAC-SHA256 vector from Python hashlib, salt bytes 0..15.
        string vector = "pbkdf2-sha256$v1$10000$AAECAwQFBgcICQoLDA0ODw==$0uRIM/qdpoCRSsIOgHbrvHycG9FmUcxf4uyuWy4hhJM=";
        Check(PasswordHasher.Verify("", "test-password", vector, false, out upgrade) && upgrade, "independent vector and work-factor upgrade");
        string[] invalid = {
            null, "", "plaintext", new string('z', 32), "pbkdf2$600000$a$b",
            stored.Replace("$v1$", "$v2$"), stored.Replace("600000", "2147483647"),
            stored.Replace("600000", "-1"), stored.Replace("600000", "0"), stored + "$extra",
            "pbkdf2-sha256$v1$600000$!!!!!!!!!!!!!!!!!!!!!!!!$" + new string('a', 44),
            "pbkdf2-sha256$v1$600000$AAECAwQFBgcICQoLDA0O$" + new string('a', 44)
        };
        foreach (string value in invalid)
        {
            Check(!PasswordHasher.Verify("staff", "secret", value, false, out upgrade) && !upgrade, "malformed hash fails closed");
        }
        Check(!PasswordHasher.Verify("", null, stored, false, out upgrade), "null password");
        // Cross-check native Windows CNG against the framework reference implementation.
        byte[] salt = new byte[16];
        for (int i = 0; i < salt.Length; i++) salt[i] = (byte)i;
        foreach (string value in new[] { "", "a\0b", "  Tiếng Việt \ud83d\udd10  " })
        {
            string reference;
            using (var kdf = new Rfc2898DeriveBytes(value, salt, 10000, HashAlgorithmName.SHA256))
                reference = "pbkdf2-sha256$v1$10000$" + Convert.ToBase64String(salt) + "$" + Convert.ToBase64String(kdf.GetBytes(32));
            Check(PasswordHasher.Verify("", value, reference, false, out upgrade) && upgrade, "CNG/framework interop including NUL");
        }
        Console.WriteLine("PASS: {0} password checks", checks);
    }
}
