using System;
using System.Data;
using Accounts.Data;

namespace Accounts.Data.SqlClient
{
    // Kept in the custom file so NetTiers regeneration retains this accessor.
    public abstract partial class SqlTaiKhoanProviderBase
    {
        protected string PasswordConnectionReference { get { return _connectionString; } }
    }


    public partial class SqlTaiKhoanProvider
    {
        public Accounts.Data.Security.PasswordStoragePolicy GetPasswordStoragePolicy()
        {
            var database = new Accounts.Data.Security.SecureSqlDatabase(PasswordConnectionReference);
            using (var command = database.GetStoredProcCommand("dbo.sp_HocPhi_PasswordPolicy_Get"))
            using (var reader = database.ExecuteReader(command))
            {
                if (!reader.Read()) throw new InvalidOperationException("Password policy has not been installed.");
                var policy = new Accounts.Data.Security.PasswordStoragePolicy
                {
                    Uis = reader.GetString(0) == "UIS",
                    Enabled = reader.GetBoolean(1),
                    MigrateOnLogin = reader.GetBoolean(2),
                    LegacyLoginDeadlineUtc = reader.IsDBNull(3) ? (DateTime?)null : reader.GetDateTime(3)
                };
                return policy;
            }
        }

        // 0 = saved, 1 = stale hash/locked/missing account. Never logs password material.
        public int CompareExchangePassword(int accountId, string expected, string replacement, TransactionManager transaction = null)
        {
            var database = new Accounts.Data.Security.SecureSqlDatabase(PasswordConnectionReference);
            using (var command = database.GetStoredProcCommand("dbo.sp_HocPhi_Password_CompareExchange"))
            {
                database.AddInParameter(command, "@MaTaiKhoan", DbType.Int32, accountId);
                database.AddInParameter(command, "@ExpectedHash", DbType.String, expected);
                database.AddInParameter(command, "@ReplacementHash", DbType.String, replacement);
                return Convert.ToInt32(transaction == null ? database.ExecuteScalar(command)
                    : database.ExecuteScalar(command, transaction.TransactionObject));
            }
        }
    }
}
