        public static byte[] DecodeBytes(string stored)
        {
            if (string.IsNullOrWhiteSpace(stored)) throw new ArgumentException("Connection configuration required.");
            string sqlPrefix = "DPAPI-SQL-LM-V1:", legacyPrefix = "DPAPI-LM-V1:";
            if (stored.StartsWith(sqlPrefix, StringComparison.Ordinal)) return Unprotect(stored.Substring(sqlPrefix.Length));
            if (stored.StartsWith(legacyPrefix, StringComparison.Ordinal))
            {
                byte[] wrapped = Unprotect(stored.Substring(legacyPrefix.Length));
                try { return DecryptCipher(Encoding.UTF8.GetChars(wrapped)); } finally { Array.Clear(wrapped, 0, wrapped.Length); }
            }
            return DecryptCipher(stored.ToCharArray());
        }
        public static SqlSession DecodeFileBytes(byte[] stored, string applicationName = null)
        {
            using (var content = DpapiConnectionFile.Read(stored))
            {
                byte[] plain = null; char[] characters = null;
                try
                {
                    if (content.Kind == DpapiContentKind.LegacyCipher)
                        plain = DecryptCipher(Encoding.UTF8.GetChars(content.Data));
                    else if (content.Kind == DpapiContentKind.SqlConnection)
                        plain = (byte[])content.Data.Clone();

            }

        private static byte[] Unprotect(string text)
        {
            byte[] cipher = Convert.FromBase64String(text);
            try { return ProtectedData.Unprotect(cipher, null, DataProtectionScope.LocalMachine); } finally { Array.Clear(cipher, 0, cipher.Length); }
        }
        private static byte[] DecryptCipher(char[] text)

                // Legacy DLL accepts plaintext SQL when a semicolon is present.
                if (Array.IndexOf(text, ';') >= 0) return Encoding.UTF8.GetBytes(text);
                for (int i = 0; i < text.Length; i++) { if (text[i] == '_') text[i] = '+'; else if (text[i] == '-') text[i] = '/'; else if (text[i] == '$') text[i] = '='; }

            finally { Array.Clear(text, 0, text.Length); if (cipher != null) Array.Clear(cipher, 0, cipher.Length); if (key != null) Array.Clear(key, 0, key.Length); }

        public static SqlSession Parse(char[] text, string applicationName = null)
        {
            var builder = new SqlConnectionStringBuilder(); string user = "";
            using (var password = new SecureString())
            {
                int position = 0;
                while (position < text.Length)
                {
                    while (position < text.Length && (char.IsWhiteSpace(text[position]) || text[position] == ';')) position++;
                    if (position == text.Length) break;
                    int begin = position;
                    while (position < text.Length && text[position] != '=' && text[position] != ';') position++;
                    if (position == text.Length || text[position] != '=') throw new ArgumentException("Invalid connection option.");
                    string name = new string(text, begin, position - begin).Trim(); position++;
                    while (position < text.Length && char.IsWhiteSpace(text[position])) position++;
                    char quote = position < text.Length && (text[position] == '\'' || text[position] == '"') ? text[position++] : '\0';
                    char[] value = new char[text.Length]; int length=0; bool closed = quote == '\0';
                    try
                    {
                        while (position < text.Length)
                        {
                            char c = text[position++];
                            if (quote != '\0' && c == quote)
                            {
                                if (position < text.Length && text[position] == quote) { position++; value[length++]=c; continue; }
                                closed = true; break;
                            }
                            if (quote == '\0' && c == ';') break; value[length++]=c;
                        }
                        if (!closed) throw new ArgumentException("Unclosed connection value.");
                        if (quote != '\0') { while (position < text.Length && char.IsWhiteSpace(text[position])) position++; if (position < text.Length && text[position] != ';') throw new ArgumentException("Invalid quoted value."); }
                        if (quote == '\0') { while (length > 0 && char.IsWhiteSpace(value[length - 1])) value[--length]='\0'; }
                        bool isPassword = name.Equals("Password", StringComparison.OrdinalIgnoreCase) || name.Equals("Pwd", StringComparison.OrdinalIgnoreCase);
                        if (isPassword) { password.Clear(); for(int index=0;index<length;index++)password.AppendChar(value[index]); }
                        else
                        {
                            char[] chars = new char[length]; Array.Copy(value,chars,length);
                            try { string option = new string(chars); if (name.Equals("User ID", StringComparison.OrdinalIgnoreCase) || name.Equals("UID", StringComparison.OrdinalIgnoreCase) || name.Equals("User", StringComparison.OrdinalIgnoreCase)) user = option; else builder[name] = option; }
                            finally { Array.Clear(chars, 0, chars.Length); }
                        }
                    }
                    finally { Array.Clear(value,0,value.Length); }
                }
                builder.PersistSecurityInfo = false;
                if (applicationName != null) builder.ApplicationName = applicationName.Length > 128 ? applicationName.Substring(0, 128) : applicationName;
                password.MakeReadOnly(); return new SqlSession(builder.ConnectionString, user, password);
            }
        }
