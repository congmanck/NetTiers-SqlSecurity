using System;
using System.Globalization;
using System.Security.Cryptography;
using System.Text;
using System.Runtime.InteropServices;

namespace __DAL_NAMESPACE__.Security
{
    /// <summary>Versioned password storage. Legacy MD5 is verification-only.</summary>
    public static class PasswordHasher
    {
        public const int Iterations = 600000;
        private const int MaximumIterations = 2000000;
        private const string Prefix = "pbkdf2-sha256$v1$";

        public static string Hash(string password)
        {
            if (password == null) throw new ArgumentNullException("password");
            byte[] salt = new byte[16];
            using (var random = RandomNumberGenerator.Create()) random.GetBytes(salt);
            return Prefix + Iterations.ToString(CultureInfo.InvariantCulture) + "$"
                + Convert.ToBase64String(salt) + "$" + Convert.ToBase64String(Derive(password, salt, Iterations));
        }

        public static bool IsModern(string stored)
        {
            int iterations;
            byte[] salt, expected;
            return TryParse(stored, out iterations, out salt, out expected);
        }

        public static bool IsLegacy(string stored)
        {
            if (stored == null || stored.Length != 32) return false;
            foreach (char c in stored)
                if (!((c >= '0' && c <= '9') || (c >= 'a' && c <= 'f') || (c >= 'A' && c <= 'F'))) return false;
            return true;
        }

        public static bool Verify(string userName, string password, string stored, bool uis, out bool needsUpgrade)
        {
            needsUpgrade = false;
            if (password == null) return false;
            int iterations;
            byte[] salt, expected;
            if (TryParse(stored, out iterations, out salt, out expected))
            {
                // UIS has not adopted this format; never reinterpret it as MD5.
                if (uis) return false;
                bool valid = FixedEquals(Derive(password, salt, iterations), expected);
                needsUpgrade = valid && iterations < Iterations;
                return valid;
            }
            if (!IsLegacy(stored)) return false;
            string material = uis
                ? "UisStaffID=" + (userName ?? "").ToUpper() + ";UisPassword=" + password.Trim()
                : "UisPassword=" + password;
            byte[] actual;
            using (var md5 = MD5.Create()) actual = md5.ComputeHash(Encoding.UTF8.GetBytes(material));
            bool matched = FixedEquals(Encoding.ASCII.GetBytes(BitConverter.ToString(actual).Replace("-", "")),
                Encoding.ASCII.GetBytes(stored.ToUpperInvariant()));
            needsUpgrade = matched && !uis;
            return matched;
        }

        private static bool TryParse(string stored, out int iterations, out byte[] salt, out byte[] expected)
        {
            iterations = 0;
            salt = expected = null;
            if (stored == null || stored.Length > 255 || !stored.StartsWith(Prefix, StringComparison.Ordinal)) return false;
            string[] fields = stored.Split('$');
            if (fields.Length != 5 || !int.TryParse(fields[2], NumberStyles.None, CultureInfo.InvariantCulture, out iterations)
                || iterations < 10000 || iterations > MaximumIterations || fields[3].Length != 24 || fields[4].Length != 44) return false;
            try
            {
                salt = Convert.FromBase64String(fields[3]);
                expected = Convert.FromBase64String(fields[4]);
                return salt.Length == 16 && expected.Length == 32
                    && Convert.ToBase64String(salt) == fields[3] && Convert.ToBase64String(expected) == fields[4];
            }
            catch (FormatException) { return false; }
        }

        private static byte[] Derive(string password, byte[] salt, int iterations)
        {
            // Windows CNG implements standard PBKDF2-HMAC-SHA256 without the
            // per-round managed overhead of .NET Framework's Rfc2898DeriveBytes.
            byte[] secret = Encoding.UTF8.GetBytes(password);
            IntPtr algorithm = IntPtr.Zero;
            try
            {
                int status = BCryptOpenAlgorithmProvider(out algorithm, "SHA256", null, 0x00000008);
                if (status < 0) throw new CryptographicException("Cannot initialize PBKDF2.");
                byte[] result = new byte[32];
                status = BCryptDeriveKeyPBKDF2(algorithm, secret, secret.Length, salt, salt.Length,
                    (ulong)iterations, result, result.Length, 0);
                if (status < 0) throw new CryptographicException("PBKDF2 computation failed.");
                return result;
            }
            finally
            {
                Array.Clear(secret, 0, secret.Length);
                if (algorithm != IntPtr.Zero) BCryptCloseAlgorithmProvider(algorithm, 0);
            }
        }

        [DllImport("bcrypt.dll", CharSet = CharSet.Unicode)]
        [DefaultDllImportSearchPaths(DllImportSearchPath.System32)]
        private static extern int BCryptOpenAlgorithmProvider(out IntPtr algorithm, string algorithmId, string implementation, int flags);

        [DllImport("bcrypt.dll")]
        [DefaultDllImportSearchPaths(DllImportSearchPath.System32)]
        private static extern int BCryptDeriveKeyPBKDF2(IntPtr algorithm, byte[] password, int passwordLength,
            byte[] salt, int saltLength, ulong iterations, [Out] byte[] result, int resultLength, int flags);

        [DllImport("bcrypt.dll")]
        [DefaultDllImportSearchPaths(DllImportSearchPath.System32)]
        private static extern int BCryptCloseAlgorithmProvider(IntPtr algorithm, int flags);

        private static bool FixedEquals(byte[] actual, byte[] expected)
        {
            if (actual.Length != expected.Length) return false;
            int difference = 0;
            for (int i = 0; i < actual.Length; i++) difference |= actual[i] ^ expected[i];
            return difference == 0;
        }
    }
}
